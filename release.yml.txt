# 打 tag 即自动发布：校验 → 打包 → 建 Release → 传 zip
#
# 触发方式（两种）：
#   1. 推一个 v* 开头的 tag：       git tag v1.0.3 && git push origin v1.0.3
#   2. 网页 Actions 页手动触发：    输入版本号即可
#
# 依赖：本工作流调用 .github/scripts/pack.py，由 enable-ci.bat 从
#       pack.py.txt 一并还原。只还原 ci.yml 而不还原 pack.py 会在这里挂掉。
#
# 为什么本仓库默认没启用：GitHub 要求 .github/workflows/ 下的文件必须用带
# Workflows: Read and write 权限的 token 推送，Contents API 会被 403。
# 所以这些配置以 .txt 形式随包提供，想启用就双击 enable-ci.bat 还原到正确位置。
#
# 注意：本工作流的入口是 "推一个 tag"。如果你的网络环境 git push 不通
# （代理拦截 git 协议），这条路径就用不上 —— 改用 release.py --version X.Y.Z
# 从本地发布，功能等价。

name: Release

on:
  push:
    tags:
      - "v*"
  workflow_dispatch:
    inputs:
      version:
        description: "版本号（不带 v，如 1.0.3）"
        required: true
        type: string

permissions:
  contents: write        # 建 Release、传附件必需

jobs:
  # 先把测试跑通，测试挂了就不发版
  verify:
    name: 校验
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"

      - name: 语法编译
        run: python -m compileall -q scripts/ tests/

      - name: 离线测试
        run: |
          python tests/test_organize.py
          python tests/test_fetch.py
          python tests/test_selfcheck.py

      - name: 隐私自检
        run: |
          set -e
          if git ls-files | grep -Ei '(^|/)(state_.*\.json|storage_state\.json|cookies.*\.json)$'; then
            echo "::error::仓库里出现登录态文件"; exit 1
          fi
          if grep -rn --include='*.py' -E 'C:\\\\Users\\\\|/Users/[a-z]|D:\\\\' scripts/; then
            echo "::error::脚本里有本机绝对路径"; exit 1
          fi
          echo "OK"

  release:
    name: 打包并发布
    needs: verify
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"

      - name: 确定版本号
        id: ver
        run: |
          if [ "${{ github.event_name }}" = "workflow_dispatch" ]; then
            V="${{ inputs.version }}"
          else
            V="${GITHUB_REF_NAME#v}"
          fi
          echo "version=$V" >> "$GITHUB_OUTPUT"
          echo "tag=v$V" >> "$GITHUB_OUTPUT"
          echo "版本: v$V"

      - name: 打包
        run: |
          test -f .github/scripts/pack.py || {
            echo "::error::缺少 .github/scripts/pack.py，请先运行 enable-ci.bat 或手动 cp pack.py.txt .github/scripts/pack.py"
            exit 1
          }
          python .github/scripts/pack.py --version "${{ steps.ver.outputs.version }}"

      - name: 发布
        uses: softprops/action-gh-release@v2
        with:
          tag_name: ${{ steps.ver.outputs.tag }}
          name: ${{ steps.ver.outputs.tag }}
          files: dist/*.zip
          generate_release_notes: true
          draft: false
          prerelease: false
