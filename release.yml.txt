# 打 tag 即自动发布：校验 → 打包 → 建 Release → 传 zip
#
# 触发方式（两种）：
#   1. 推一个 v* 开头的 tag：       git tag v1.0.3 && git push origin v1.0.3
#   2. 网页 Actions 页手动触发：    输入版本号即可
#
# 为什么本仓库可能用不上：GitHub 要求 .github/workflows/ 下的文件必须用带
# Workflows 权限的 token 推送，Contents API 会被 403。如果这个文件没在仓库里，
# 把 ci.yml.txt 旁边的这份 release-notes.yml 按同样方式启用（见 enable-ci.bat 说明）。
#
# 若本机 git 协议不通（代理拦截），可以改用 release.py --version X.Y.Z 从本地发。

name: Release

on:
  push:
    tags:
      - "v*"
  workflow_dispatch:
    inputs:
      version:
        description: "版本号（不带 v，如 1.0.3）"
        required: true
        type: string

permissions:
  contents: write        # 建 Release、传附件必需

jobs:
  # 先把测试跑通，测试挂了就不发版
  verify:
    name: 校验
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"

      - name: 语法编译
        run: python -m compileall -q scripts/ tests/

      - name: 离线测试
        run: |
          python tests/test_organize.py
          python tests/test_fetch.py

      - name: 隐私自检
        run: |
          set -e
          if git ls-files | grep -Ei '(^|/)(state_.*\.json|storage_state\.json|cookies.*\.json)$'; then
            echo "::error::仓库里出现登录态文件"; exit 1
          fi
          if grep -rn --include='*.py' -E 'C:\\\\Users\\\\|/Users/[a-z]|D:\\\\' scripts/; then
            echo "::error::脚本里有本机绝对路径"; exit 1
          fi
          echo "OK"

  release:
    name: 打包并发布
    needs: verify
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"

      - name: 确定版本号
        id: ver
        run: |
          if [ "${{ github.event_name }}" = "workflow_dispatch" ]; then
            V="${{ inputs.version }}"
          else
            V="${GITHUB_REF_NAME#v}"
          fi
          echo "version=$V" >> "$GITHUB_OUTPUT"
          echo "tag=v$V" >> "$GITHUB_OUTPUT"
          echo "版本: v$V"

      - name: 打包
        run: python .github/scripts/pack.py --version "${{ steps.ver.outputs.version }}"

      - name: 发布
        uses: softprops/action-gh-release@v2
        with:
          tag_name: ${{ steps.ver.outputs.tag }}
          name: ${{ steps.ver.outputs.tag }}
          files: dist/*.zip
          generate_release_notes: true
          draft: false
          prerelease: false
